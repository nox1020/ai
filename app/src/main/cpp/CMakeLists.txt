cmake_minimum_required(VERSION 3.31.6)
project(ogba_jni C CXX)
set(CMAKE_CXX_STANDARD 17)

# Shared ggml/llama + dynamically loaded backends:
#  - CPU: one variant per Arm feature level (armv8.0 / 8.2 dotprod / 8.2 fp16 / 8.6 i8mm / 9.x SVE2/SME),
#    picked at runtime by ggml's CPU feature scoring.
#  - Vulkan GPU backend (optional, OGBA_VULKAN), falls back to CPU if no usable device.
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(GGML_SYSTEM_ARCH "ARM" CACHE STRING "" FORCE)
  set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
  if(OGBA_VULKAN)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
  endif()
else()
  set(GGML_SYSTEM_ARCH "x86" CACHE STRING "" FORCE)
  set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(llama.cpp build-llama)
add_library(ogba_jni SHARED llama_jni.cpp)
target_include_directories(ogba_jni PRIVATE llama.cpp/include llama.cpp/ggml/include)
target_link_libraries(ogba_jni llama ggml ggml-base log android)
